## 四、duration 自动反算脚本（可并入 main.js）

```js
function syncMarquee(track, speedPxPerSec) {
  if (!track) return;
  const half = track.scrollWidth / 2;              // 一份内容宽（需已复制两份且未被拉伸）
  if (!half) return;
  track.style.animationDuration = (half / speedPxPerSec).toFixed(2) + 's';
}

function syncAllMarquees() {
  syncMarquee(document.querySelector('.ann-ticker'), 12);
  syncMarquee(document.querySelector('.nb-ticker-track'), 8);
  syncMarquee(document.querySelector('.hero-ticker-track'), 40);
}

document.fonts.ready.then(syncAllMarquees);
window.addEventListener('resize', syncAllMarquees);
```
