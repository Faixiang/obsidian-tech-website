## 四、动态重建 + 重算脚本（可并入 main.js）

```js
// 一条轨道重建内容后，重算 duration 并重置动画，避免位移基准突变
function rebuildTicker(track, html, speedPxPerSec) {
  if (!track) return;
  track.innerHTML = html + html;                 // 必须仍是两份
  const half = track.scrollWidth / 2;            // 一份内容宽
  if (!half) return;
  track.style.animation = 'none';                // 先停掉旧动画
  void track.offsetWidth;                        // 强制回流，清掉旧位移
  track.style.animationName = track.dataset.kf;  // marquee-ann / marquee-nb / marquee-hero
  track.style.animationDuration = (half / speedPxPerSec).toFixed(2) + 's';
  track.style.animationTimingFunction = 'linear';
}

// 切频道示例
function switchChannel(html) {
  rebuildTicker(document.querySelector('.nb-ticker-track'), html, 8);
}
```

> 注意：`track.dataset.kf` 需在 HTML 上声明，如 `<div class="nb-ticker-track" data-kf="marquee-nb">`。
